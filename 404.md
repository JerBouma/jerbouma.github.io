<script>
// Links to this site are often pasted or rendered with the sentence's
// punctuation stuck to the end, e.g. "/fmp." or "/docs/historical-data)."
// (from plain-text README and docstring links). Strip it and go to the page.
(function () {
  var path = decodeURIComponent(location.pathname);
  var clean = path.replace(/[).,;:!?'"\]>]+$/, '');
  if (clean !== path && clean.length > 1) {
    location.replace(clean + location.search + location.hash);
  }
})();
</script>

<script>
// Suggest pages from the sitemap whose address shares words with the one
// that wasn't found (/docs/sharpe-ratios finds /docs/performance/sharpe-ratio).
(function () {
  var words = decodeURIComponent(location.pathname).toLowerCase().split(/[^a-z0-9]+/)
    .filter(function (w) { return w.length > 2 && ['projects', 'docs', 'html', 'financetoolkit', 'financedatabase'].indexOf(w) < 0; })
    .map(function (w) { return w.length > 4 ? w.replace(/s$/, '') : w; }); // ratios finds ratio
  if (!words.length || !window.fetch) return;
  fetch('/sitemap.xml').then(function (r) { return r.text(); }).then(function (xml) {
    var urls = (xml.match(/<loc>[^<]+<\/loc>/g) || []).map(function (l) { return l.slice(5, -6); });
    var scored = urls.map(function (u) {
      var path = u.replace(/^https?:\/\/[^/]+/, '').toLowerCase();
      var last = path.split('/').pop();
      var score = words.reduce(function (n, w) { return n + (last.indexOf(w) >= 0 ? 2 : path.indexOf(w) >= 0 ? 1 : 0); }, 0);
      return { url: path || '/', score: score };
    }).filter(function (x) { return x.score > 0; })
      .sort(function (a, b) { return b.score - a.score || a.url.length - b.url.length; }).slice(0, 5);
    if (!scored.length) return;
    var box = document.getElementById('error-guess'), list = box.querySelector('ul');
    scored.forEach(function (x) {
      var li = document.createElement('li'), a = document.createElement('a');
      a.href = x.url; a.textContent = x.url; li.appendChild(a); list.appendChild(li);
    });
    box.hidden = false;
  }).catch(function () {});
})();
</script>
