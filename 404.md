<script>
// Links to this site are often pasted or rendered with the sentence's
// punctuation stuck to the end, e.g. "/fmp." or "/docs/historical-data)."
// (from plain-text README and docstring links). Strip it and go to the page.
(function () {
  // or with text from a search snippet glued on, e.g. "/articles/binomial-trees:;<b>Binomial":
  // nothing on this site has ":", ";" or "<" in its address, so cut from there
  var path = decodeURIComponent(location.pathname);
  var clean = path.replace(/[:;<].*$/, '').replace(/[).,;:!?'"\]>]+$/, '');
  if (clean !== path && clean.length > 1) {
    location.replace(clean + location.search + location.hash);
  }
})();
</script>

<script>
// Suggest pages from the search index whose address shares words with the one
// that wasn't found (/docs/sharpe-ratios finds the Sharpe Ratio page), by title.
(function () {
  var words = decodeURIComponent(location.pathname).toLowerCase().split(/[^a-z0-9]+/)
    .filter(function (w) { return w.length > 2 && ['projects', 'docs', 'html', 'financetoolkit', 'financedatabase'].indexOf(w) < 0; })
    .map(function (w) { return w.length > 4 ? w.replace(/s$/, '') : w; }); // ratios finds ratio
  if (!words.length || !window.fetch) return;
  fetch('/assets/data/search.json').then(function (r) { return r.json(); }).then(function (pages) {
    var scored = pages.map(function (p) {
      var path = p.u.toLowerCase();
      var last = path.split('/').pop();
      var score = words.reduce(function (n, w) { return n + (last.indexOf(w) >= 0 ? 2 : path.indexOf(w) >= 0 ? 1 : 0); }, 0);
      return { url: p.u || '/', title: p.t, score: score };
    }).filter(function (x) { return x.score > 0; })
      .sort(function (a, b) { return b.score - a.score || a.url.length - b.url.length; }).slice(0, 6);
    if (!scored.length) return;
    var box = document.getElementById('error-guess'), list = box.querySelector('ul');
    scored.forEach(function (x) {
      var li = document.createElement('li'), a = document.createElement('a');
      a.href = x.url; a.textContent = x.title; li.appendChild(a); list.appendChild(li);
    });
    box.hidden = false;
  }).catch(function () {});
})();
</script>

<script>
// The search opens with the words of the address that wasn't found.
document.getElementById('error-search').addEventListener('click', function () {
  var words = decodeURIComponent(location.pathname).split(/[^A-Za-z0-9]+/)
    .filter(function (w) { return w.length > 2 && ['projects', 'docs', 'html', 'financetoolkit', 'financedatabase', 'articles'].indexOf(w.toLowerCase()) < 0; });
  if (window.siteSearch) window.siteSearch(words.join(' '));
});
</script>
